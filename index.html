function showEditProfitModal() { const profitInput = document.getElementById('profitInput'); profitInput.value = ''; profitInput.type = 'number'; document.getElementById('editProfitModal').style.display = 'flex'; profitInput.focus(); profitInput.select(); } function hideEditProfitModal() { document.getElementById('editProfitModal').style.display = 'none'; } function saveProfitAmount() { const inputVal = document.getElementById('profitInput').value.trim(); if (!inputVal) { showNotification("❌ Please enter an amount"); return; } const val = money(parseFloat(inputVal)); if (isNaN(val)) { showNotification("❌ Invalid amount"); return; } if (val < 0) { showNotification("❌ Amount cannot be negative"); return; } baseProfit = val; saveData(); renderExpenses(); hideEditProfitModal(); showNotification(`✅ Base Profit updated to ₱${val}`); addLog(`Set base profit to ₱${val}`); triggerAutoDownload(); }
